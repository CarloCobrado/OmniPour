@startuml
skinparam activity {
    BackgroundColor #f8cecc
    BorderColor #b85450
    FontColor #000000
    FontSize 11
}
skinparam diamond {
    BackgroundColor #fff2cc
    BorderColor #d6b656
    FontColor #000000
    FontSize 10
}

start
:Log in to BEANTRACK;
:Verify Credentials & Role;

if (Valid User?) then (yes)
    :Displays Dashboard Actions Based on the Role;
    
    switch (Select Activity)
    case (Cashier)
        :Cashier Enters Customer Account & Order Items;
        :Auto-Apply Pricing Tier & Process Payment Details;
        :Save Sales Transaction & Issue Official Receipt;
        :Deduct Packaged Stock or Recipe Raw Ingredients;
    case (Stock Staff)
        :Stock Staff Receives Inbound Supplier Shipment;
        :Record P.O. Number, Cases & Unit Quantities;
        :Increment Target Branch Stock Balances;
        :Log Damaged / Expired Stock with Reason & Deduct Loss;
    case (Supervisor)
        :Supervisor and Stock Staff Creates New Menu Items;
        :Formulate Drink Recipes Linked to Raw Ingredients;
        :Supervisor Creates Operational Reports for Manager;
        :Export Verified 8-Sheet Client Excel (.xlsx) / CSV;
    case (Management)
        :Manager Accesses Executive Dashboard;
        :Manager Views Reports Sent by the Supervisor;
        :Review Real-Time Sales KPIs & Inventory Trends;
        :Manager Can Export the Report to PDF or Excel;
    case (System Admin)
        :Admin Creates and Manage Users & Roles;
        :Manage User Access & Operational Permissions;
        :Admin Can Save and Restore Backups;
        :Generate Database Archive or Restore Tables;
    case (Audit Trail)
        :Supervisor and Admin Has Access to Blockchain Audit Log;
        :Audit Logs Can Be Verified if Tampered Through Hash;
        :Export Immutable SHA-256 Checkpoint Snapshot;
        :Compare Checkpoints to Detect Unauthorized Modifications;
    endswitch

    ' Stock threshold evaluation for inventory/sales
    :Check Inventory Levels;
    if (Stock Below Threshold?) then (yes)
        :Flags Low Stock Items and Notify User;
    else (no)
        :Maintain Normal In-Stock Status;
    endif

    :Changes Saved in the Centralized Database;
    :User Activity gets recorded in the Audit Logs;
    :Update Reports and Dashboard;
    :Create or Verify Audit Checkpoint;

    if (Passed Check?) then (yes)
        :Backup and Securely Store Database;
    else (no)
        :Check Previous Audit Logs;
        :Review Record Differences;
    endif

    :Log out;
    stop
else (no)
    :Show Error and Retry;
    detach
endif
@enduml
