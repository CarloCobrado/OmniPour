        C_Receipt["Print / Issue Receipt"] --> C_End(["End"])
    end

        S_Normal["Maintain Normal In-Stock Status"]

        S_Compile["Retrieve Real-Time Sales, Deliveries, Spoilage & Stock"]
        S_GenRep["Generate Dynamic Multi-Sheet Workbook (.xlsx)"]
    end

        I_Action{"Operational Task?"}

    end

        M_Login --> M_Dash["Review Dashboard KPIs & Sales Summaries"]
        M_Dash --> M_Review["Inspect Submitted Multi-Sheet Reports"] --> M_End(["End"])
