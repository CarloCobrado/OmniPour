@startuml
skinparam roundcorner 5
skinparam defaultFontName Helvetica
skinparam defaultFontSize 10

skinparam class {
    BackgroundColor #ffffff
    BorderColor #b85450
    HeaderBackgroundColor #f8cecc
    FontColor #000000
    FontSize 11
    FontStyle bold
}

entity "ROLES" as Roles {
    * id : INT <<PK>>
    --
    role_name : VARCHAR(50)
}

entity "USERS" as Users {
    * id : INT <<PK>>
    --
    * role_id : INT <<FK>>
    name : VARCHAR(100)
    email : VARCHAR(120)
    password : VARCHAR(255)
    is_active : TINYINT
    created_at : TIMESTAMP
}

entity "CUSTOMERS" as Customers {
    * id : INT <<PK>>
    --
    name : VARCHAR(50)
    default_price_type : ENUM
    contact_no : VARCHAR(50)
    address : VARCHAR(255)
    active : TINYINT
    created_at : TIMESTAMP
}

entity "BRANCHES" as Branches {
    * id : INT <<PK>>
    --
    name : VARCHAR(100)
    code : VARCHAR(20)
    address : VARCHAR(255)
    created_at : TIMESTAMP
}

entity "TRANSACTIONS" as Transactions {
    * id : INT <<PK>>
    --
    * cashier_id : INT <<FK>>
    * customer_id : INT <<FK>>
    * branch_id : INT <<FK>>
    total_amount : DECIMAL(10,2)
    payment_method : ENUM
    payment_status : ENUM
    created_at : TIMESTAMP
}

entity "TRANSACTION_ITEMS" as TransactionItems {
    * id : INT <<PK>>
    --
    * transaction_id : INT <<FK>>
    * product_id : INT <<FK>>
    quantity : INT
    unit_price : DECIMAL(10,2)
    subtotal : DECIMAL(10,2)
}

entity "PRODUCTS" as Products {
    * id : INT <<PK>>
    --
    brand : VARCHAR(80)
    name : VARCHAR(120)
    category : VARCHAR(80)
    qty_per_case : INT
    unit : VARCHAR(20)
    srp : DECIMAL(10,2)
    wholesale_price : DECIMAL(10,2)
    bulk_price : DECIMAL(10,2)
    dealer_price : DECIMAL(10,2)
    reseller_price : DECIMAL(10,2)
    quantity : INT
    reorder_level : INT
    active : TINYINT
}

entity "PRODUCT_RECIPE_ITEMS" as ProductRecipeItems {
    * product_id : INT <<PK, FK>>
    * ingredient_id : INT <<PK, FK>>
    --
    quantity_required : DECIMAL(12,3)
}

entity "INGREDIENTS" as Ingredients {
    * id : INT <<PK>>
    --
    name : VARCHAR(120)
    unit : VARCHAR(30)
    quantity : DECIMAL(12,3)
    reorder_level : DECIMAL(12,3)
    active : TINYINT
    created_at : TIMESTAMP
}

entity "SUPPLIER_DELIVERIES" as SupplierDeliveries {
    * id : INT <<PK>>
    --
    * branch_id : INT <<FK>>
    * product_id : INT <<FK>>
    * received_by : INT <<FK>>
    date_arrived : DATE
    supplier : VARCHAR(120)
    po_no : VARCHAR(60)
    cases : DECIMAL(10,2)
    qty : DECIMAL(10,2)
    remarks : TEXT
}

entity "INVENTORY_SPOILAGE" as InventorySpoilage {
    * id : INT <<PK>>
    --
    * branch_id : INT <<FK>>
    * product_id : INT <<FK>>
    * recorded_by : INT <<FK>>
    date : DATE
    qty : DECIMAL(10,2)
    remarks : TEXT
}

entity "AUDIT_TRAIL" as AuditTrail {
    * id : INT <<PK>>
    --
    * user_id : INT <<FK>>
    action : VARCHAR(120)
    table_name : VARCHAR(80)
    record_id : INT
    description : TEXT
    previous_hash : CHAR(64)
    entry_hash : CHAR(64)
    created_at : TIMESTAMP
}

entity "AUDIT_CHAIN_STATE" as AuditChainState {
    * id : TINYINT <<PK>>
    --
    * last_id : INT <<FK>>
    * chain_start_id : INT <<FK>>
    last_hash : CHAR(64)
}

' Relationships
Roles ||--o{ Users : "assigned to"
Users ||--o{ Transactions : "processes"
Customers ||--o{ Transactions : "places"
Branches ||--o{ Transactions : "houses"
Transactions ||--|{ TransactionItems : "contains"
Products ||--o{ TransactionItems : "ordered in"
Products ||--o{ ProductRecipeItems : "composed of"
Ingredients ||--o{ ProductRecipeItems : "used in"
Products ||--o{ SupplierDeliveries : "received in"
Products ||--o{ InventorySpoilage : "damaged in"
Users ||--o{ AuditTrail : "triggers"
AuditTrail ||--|| AuditChainState : "tracks latest head"
Branches ||--o{ InventorySpoilage : "branch"

@enduml
